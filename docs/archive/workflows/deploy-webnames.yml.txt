# ARCHIVED 2026-08-18 — DO NOT RESTORE AS AN ACTIVE WORKFLOW.
#
# This workflow targeted the obsolete Webnames hosting configuration, always
# checked out `main`, and uploaded the repository root. It never performed a
# confirmed production deployment. It is retained only as an audit record.

name: Deploy to Webnames (manual)

on:
  workflow_dispatch:
    inputs:
      dry_run:
        description: "Only show the planned changes; do not upload files"
        required: true
        default: true
        type: boolean

run-name: "Webnames: ${{ github.ref_name }}"

permissions:
  contents: read

concurrency:
  group: webnames-egoe-production
  cancel-in-progress: false

jobs:
  deploy:
    name: Deploy static site
    runs-on: ubuntu-latest
    timeout-minutes: 45
    environment:
      name: webnames-production
    steps:
      - name: Check out production source
        uses: actions/checkout@v4
        with:
          ref: main
          fetch-depth: 1

      - name: Deploy over FTP or FTPS
        uses: SamKirkland/FTP-Deploy-Action@v4.4.0
        with:
          server: ${{ vars.WEBNAMES_FTP_HOST }}
          username: ${{ vars.WEBNAMES_FTP_USERNAME }}
          password: ${{ secrets.WEBNAMES_FTP_PASSWORD }}
          protocol: ${{ vars.WEBNAMES_FTP_PROTOCOL }}
          port: ${{ vars.WEBNAMES_FTP_PORT }}
          security: loose
          local-dir: ./
          server-dir: ./
          dangerous-clean-slate: false
          dry-run: ${{ inputs.dry_run }}
          log-level: standard
          timeout: 60000
          exclude: |
            **/.git*
            **/.git*/**
            **/.github/**
            **/.claude/**
            **/node_modules/**
            **/.DS_Store
            **/Thumbs.db
            **/.env
            **/.env.*
            **/*.log
            **/*.md
            .gitignore
            tools/*.mjs
            CNAME
            .nojekyll
